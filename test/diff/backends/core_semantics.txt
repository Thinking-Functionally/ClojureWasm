;; Three-way differential cases (tree_walk / vm / wasmgc).
;; Cases are separated by a line containing exactly `;;---`. Each case is run as
;; `(do <case>)` in a separate cljw process per backend; scripts/check_backend_diff.sh
;; requires identical stdout, exit code and error header across all three.
;; Keep cases deterministic (no clock/random/unordered output).

;; ---- closures and captures
(let [x 10] ((fn [y] (+ x y)) 5))
;;---
(vec (map (fn [n] ((fn [x] (+ x n)) 10)) [1 2 3]))
;;---
(let [c (atom 0) f (fn [] (swap! c inc))] (f) (f) (f))
;;---
(letfn [(ev? [n] (if (zero? n) true (od? (dec n)))) (od? [n] (if (zero? n) false (ev? (dec n))))] [(ev? 10) (od? 7) (ev? 7)])
;;---
((fn fact [n] (if (< n 2) 1 (* n (fact (dec n))))) 12)
;;---
((((fn [a] (fn [b] (fn [c] (+ a b c)))) 1) 2) 3)
;;---
(let [fs (map (fn [i] (fn [] i)) (range 4))] (mapv (fn [f] (f)) fs))
;;---
(let [compose (fn [f g] (fn [x] (f (g x))))] ((compose inc (fn [x] (* x 2))) 5))

;; ---- arities and varargs
;;---
(let [f (fn ([] :zero) ([a] :one) ([a b] :two) ([a b & r] [:many r]))] [(f) (f 1) (f 1 2) (f 1 2 3 4)])
;;---
(apply (fn [a & r] [a r]) 1 2 [3 4])
;;---
((fn [& xs] (count xs)))
;;---
((fn [a b] a) 1)

;; ---- loop / recur
;;---
(loop [i 0 acc []] (if (< i 5) (recur (inc i) (conj acc (* i i))) acc))
;;---
((fn [n acc] (if (zero? n) acc (recur (dec n) (+ acc n)))) 1000 0)
;;---
(loop [i 0] (if (< i 100000) (recur (inc i)) i))
;;---
(loop [a 1 b 2] (if (> a 50) [a b] (recur (+ a b) a)))
;;---
(loop [xs (range 5) out []] (if (seq xs) (recur (rest xs) (conj out (first xs))) out))

;;---

;;---

;;---

;;---
