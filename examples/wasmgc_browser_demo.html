Sample Clojure Code

;; Example 1: Console logging
(defn hello-console []
  (js/console.log "Hello from ClojureWasm!"))

;; Example 2: DOM manipulation
(defn update-dom [selector text]
  (let [elem (js/document.querySelector selector)]
    (set! (.-textContent elem) text)))

;; Example 3: Exported function
(defn ^:export fibonacci [n]
  (loop [a 0 b 1 i n]
    (if (= i 0)
      a
      (recur b (+ a b) (dec i)))))

;; Example 4: String processing
(defn ^:export greet [name]
  (str "Hello, " name "! Welcome to ClojureWasm."))

Generated WAT Module Structure

(module
  ;; Host imports
  (import "console" "log" (func $console_log (param externref)))
  (import "dom" "querySelector" (func $dom_querySelector
    (param externref) (result externref)))
  (import "dom" "setTextContent" (func $dom_setTextContent
    (param externref externref)))

  ;; Type definitions
  (type $Value (struct
    (field $tag i32)
    (field $data (ref null $ValueData))))

  ;; Exported functions
  (func $fibonacci (export "fibonacci") (param $n i64) (result i64)
    ;; Implementation...
  )

  (func $greet (export "greet") (param $name (ref $Value))
    (result (ref $Value))
    ;; Implementation...
  )

  ;; Conversion helpers
  (func $clojure_to_js (param (ref $Value)) (result externref)
    ;; Type-based conversion
  )

  (func $js_to_clojure (param externref) (result (ref $Value))
    ;; JS typeof checking and conversion
  )
)

Browser Requirements

Testing with wasmtime (CLI)

# Compile WAT to WASM
wasm-tools parse clojure.wat -o clojure.wasm

# Run with wasmtime
wasmtime run clojure.wasm --invoke fibonacci 10

# Validate module
wasm-tools validate clojure.wasm
